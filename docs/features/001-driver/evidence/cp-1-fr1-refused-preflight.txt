## What the passing test asserts (tests/run.rs:247-262)
A fixture with one uncommitted file, then `factory run 001` through the binary with the stub claude/gh on PATH:
  exit 1 · stdout holds the tool's own refuse text (`✗ working tree not clean (1 path(s))`, `refused: 1 blocker(s)`)
  no runs/ entry under FACTORY_DATA_DIR · calls/ empty (no stub claude spawn) · HEAD still main (no branch created)

#[test]
fn run_refuses_a_dirty_tree_with_no_run_dir_and_no_spawn() {
    let fx = Fixture::new("run-dirty");
    fs::write(fx.path().join("dirty.txt"), "x\n").unwrap();
    let out = factory(&fx, &["run", "001"]);
    assert_eq!(out.status.code(), Some(1), "{}", stdout(&out));
    let text = stdout(&out);
    assert!(
        text.contains("✗ working tree not clean (1 path(s))"),
        "{text}"
    );
    assert!(text.contains("refused: 1 blocker(s)"), "{text}");
    assert!(run_dirs(&fx).is_empty());
    assert!(fx.calls().is_empty());
    assert_eq!(head(&fx), "main");
}
