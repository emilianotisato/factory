## Script

    #!/usr/bin/env bash
    # Times `factory ctx` verbs: one warm-up run, then three timed runs each.
    F=$CARGO_TARGET_DIR/release/factory
    S=$(dirname "$0")
    TIMEFORMAT='real %3R s  user %3U s  sys %3S s'
    row() { # repo-dir, args...
      local dir=$1; shift
      cd "$dir" || { echo "MISSING $dir"; return; }
      echo "\$ cd $dir && factory ctx $*"
      "$F" ctx "$@" > "$S/warm.out" 2> "$S/warm.err"; local code=$?
      echo "warm-up: exit=$code, stdout $(wc -l < "$S/warm.out") lines / $(wc -c < "$S/warm.out") bytes; first line: $(head -1 "$S/warm.out")"
      echo "         last line: $(tail -1 "$S/warm.out")"
      [ -s "$S/warm.err" ] && echo "         stderr: $(head -2 "$S/warm.err")"
      for i in 1 2 3; do
        { time "$F" ctx "$@" > /dev/null 2>&1; } 2> "$S/t.txt"
        echo "run $i: $(cat "$S/t.txt")"
      done
      echo
    }
    echo "binary: $F ($(stat -c %s "$F") bytes, built $(stat -c %y "$F" | cut -d. -f1)), cargo build --release at factory $(git -C /home/emiliano/Projects/factory rev-parse --short HEAD)"
    echo "machine: $(nproc) cores, $(uname -sr)"
    echo
    I=/home/emiliano/Projects/ICG/icg-cloud
    A=/home/emiliano/Projects/Prokey/arca-api
    row $I def Company
    row $I refs Company
    row $I outline app/Core/Http/Controllers/Admin
    row $I outline app/Services
    row $A def Code
    row $A refs Code
    row $A outline internal/caea
    row $A outline internal/
